# Modelo de variáveis de ambiente para produção (VPS).
#
# Copia para .env.production e preenche os valores reais — esse ficheiro
# NUNCA é commitado (ver .gitignore). É lido por docker-compose.prod.yml.
#
#   cp .env.production.example .env.production

# --- Servidor ---
NODE_ENV=production
PORT=3333

# --- Base de Dados PostgreSQL ---
# DB_HOST tem de ser "postgres" (nome do serviço no docker-compose.prod.yml,
# resolvido pela rede interna do Docker) — nunca "localhost" em produção.
DB_HOST=postgres
DB_PORT=5432
DB_NAME=kuava_pos
DB_USER=kuava
# Gera uma senha forte e única, ex.: openssl rand -base64 24
DB_PASSWORD=GERAR_SENHA_FORTE_AQUI
DB_LOGGING=false

# --- Autenticação ---
# Obrigatório e tem de ser forte — a app recusa arrancar em produção sem
# isto (ver kuava-api/src/config/env.ts). Gera com: openssl rand -hex 32
JWT_SECRET=GERAR_SEGREDO_FORTE_AQUI
JWT_EXPIRES_IN=8h

# --- CORS ---
# O domínio real da app em produção (o Nginx serve frontend e API do mesmo
# domínio, mas mantém-se como defesa extra contra acesso direto à API).
CORS_ORIGIN=https://kuava.vektramz.com

# --- Regras de Negócio ---
IVA_RATE=0.16

# --- Frontend (build-time, ver kuava-web/Dockerfile) ---
# "/api" porque o Nginx serve o frontend e faz proxy de /api no mesmo
# domínio — assim o browser nunca faz pedidos cross-origin.
VITE_API_BASE_URL=/api

# --- Nginx / TLS (ver deploy/README.md) ---
DOMAIN=kuava.vektramz.com

# --- Backup do Postgres (ver deploy/README.md, secção "Backup do Postgres") ---
# Pasta no VPS onde os backups ficam guardados (fora do repositório de
# propósito). Cria-a com permissões adequadas antes do primeiro backup, ex.:
#   sudo mkdir -p /var/backups/kuava-postgres && sudo chown $USER /var/backups/kuava-postgres
BACKUP_DIR=/var/backups/kuava-postgres
# Quantos dias de backups LOCAIS manter (os mais antigos são apagados
# automaticamente a cada corrida do script) — o envio externo abaixo, se
# ativado, nunca é apagado por este script.
BACKUP_RETENTION_DAYS=14

# Envio externo opcional para um bucket S3-compatível (AWS S3, Backblaze B2,
# etc.) — deixa BACKUP_S3_BUCKET vazio para desativar e manter só backups
# locais. Se ativares, usa credenciais com permissão APENAS de escrita
# (PutObject), nunca de apagar (DeleteObject) — assim, mesmo que o VPS seja
# comprometido, quem lá entrar não consegue apagar os teus backups já
# enviados, só criar novos.
BACKUP_S3_BUCKET=
# Só necessário para fornecedores S3-compatíveis que não sejam a AWS (ex.
# Backblaze B2: https://s3.REGIAO.backblazeb2.com). Deixa vazio para AWS S3.
BACKUP_S3_ENDPOINT=
BACKUP_S3_REGION=us-east-1
BACKUP_S3_PREFIX=kuava-postgres
BACKUP_S3_ACCESS_KEY_ID=
BACKUP_S3_SECRET_ACCESS_KEY=
